## 5. 헬스체크 엔드포인트 (신규 생성 필요)

아래 코드를 `app/api/debug/health/route.ts`에 생성할 것.

```typescript
// app/api/debug/health/route.ts
export async function GET() {
  const checks: Record<string, boolean> = {};
  
  // CoreRing 테스트
  try {
    const res = await fetch('http://localhost:3000/api/brainpool', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json; charset=utf-8' },
      body: JSON.stringify({ text: 'test' })
    });
    checks.coreRing = res.ok;
  } catch { checks.coreRing = false; }

  // DB 연결 테스트 (간단 쿼리)
  try {
    const { createClient } = await import('@supabase/supabase-js');
    const supabase = createClient(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.SUPABASE_SERVICE_ROLE_KEY!
    );
    const { error } = await supabase.from('chat_rooms').select('id', { count: 'exact', head: true });
    checks.db = !error;
  } catch { checks.db = false; }

  return Response.json({
    status: Object.values(checks).every(Boolean) ? 'ok' : 'degraded',
    checks,
    timestamp: new Date().toISOString()
  });
}
```
